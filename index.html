// ==================================================================== // RECEPTOR MQTT (HIVEMQ) - LILYGO T-PCIE (SIM7600 / TELCEL) // ==================================================================== #define TINY_GSM_MODEM_SIM7600 #include #include // --- PINES DE LA LILYGO T-PCIE PARA EL MÓDEM SIM7600 --- #define MODEM_TX 27 #define MODEM_RX 26 #define MODEM_PWRKEY 4 #define MODEM_POWER_ON 25 // --- CONFIGURACIÓN DE APN TELCEL MÉXICO --- const char apn[] = "internet.itelcel.com"; const char gprsUser[] = "webgprs"; const char gprsPass[] = "webgprs2002"; // --- CREDENCIALES DE HIVEMQ (Extraídas de tu HTML) --- const char* mqtt_server = "810eb95b7b3f456990ad8fce9280f402.s1.eu.hivemq.cloud"; const int mqtt_port = 1883; // Puerto TCP estándar para clientes de hardware const char* mqtt_user = "sp32"; const char* mqtt_pass = "123456789"; const char* mqtt_topic = "cnc/control"; HardwareSerial SerialModem(1); TinyGsm modem(SerialModem); TinyGsmClient client(modem); PubSubClient mqtt(client); void encenderModem() { pinMode(MODEM_POWER_ON, OUTPUT); digitalWrite(MODEM_POWER_ON, HIGH); pinMode(MODEM_PWRKEY, OUTPUT); digitalWrite(MODEM_PWRKEY, LOW); delay(100); digitalWrite(MODEM_PWRKEY, HIGH); delay(1000); digitalWrite(MODEM_PWRKEY, LOW); delay(3000); } // Función que se ejecuta cada vez que llega una orden desde la página web void callbackMQTT(char* topic, byte* payload, unsigned int length) { Serial.print("\n[MQTT] ¡Orden recibida en el tópico: "); Serial.print(topic); Serial.println("]"); String mensaje = ""; for (unsigned int i = 0; i < length; i++) { mensaje += (char)payload[i]; } Serial.print("[Comando] Valor: "); Serial.println(mensaje); // Lógica para controlar tu sistema (FarmBot / Actuador) if (mensaje == "1") { Serial.println("-> Acciones: ¡ENCENDER sistema / riego / motor!"); // Aquí puedes activar un pin digital de tu ESP32: digitalWrite(PIN_RELAY, HIGH); } else if (mensaje == "0") { Serial.println("-> Acciones: ¡APAGAR sistema / riego / motor!"); // Aquí puedes desactivar el pin: digitalWrite(PIN_RELAY, LOW); } } void reconectarMQTT() { while (!mqtt.connected()) { Serial.print("Conectando a HiveMQ Cloud vía Telcel..."); // Generar un ID de cliente único String clientId = "LilyGO-FarmBot-"; clientId += String(random(0xffff), HEX); if (mqtt.connect(clientId.c_str(), mqtt_user, mqtt_pass)) { Serial.println(" ¡Conectado con éxito!"); // Suscribirse al mismo tópico que usa tu página web mqtt.subscribe(mqtt_topic); } else { Serial.print(" Falló, rc="); Serial.print(mqtt.state()); Serial.println(" Reintentando en 5 segundos..."); delay(5000); } } } void setup() { Serial.begin(115200); delay(1000); Serial.println("\n========================================="); Serial.println("INICIANDO SISTEMA MQTT 4G - TELCEL"); Serial.println("========================================="); encenderModem(); SerialModem.begin(115200, SERIAL_8N1, MODEM_RX, MODEM_TX); delay(3000); Serial.println("Reiniciando módem SIM7600..."); modem.restart(); Serial.println("Buscando señal de la torre Telcel..."); if (!modem.waitForNetwork(60000)) { Serial.println("[ERROR] Sin señal celular."); return; } Serial.println("[OK] ¡Red celular encontrada!"); Serial.print("Conectando al APN de Telcel ("); Serial.print(apn); Serial.println(")..."); if (!modem.gprsConnect(apn, gprsUser, gprsPass)) { Serial.println("[ERROR] No se pudo establecer conexión GPRS."); return; } Serial.println("[OK] ¡Conectado a los datos móviles de Telcel!"); // Configurar servidor MQTT mqtt.setServer(mqtt_server, mqtt_port); mqtt.setCallback(callbackMQTT); } void loop() { // Mantener viva la conexión MQTT if (!mqtt.connected()) { reconectarMQTT(); } mqtt.loop(); }
